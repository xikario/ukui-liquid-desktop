cmake_minimum_required(VERSION 3.16)
project(UkuiLiquidPopup LANGUAGES CXX)
find_package(Qt5 REQUIRED COMPONENTS Core Gui Widgets)
add_library(ukui-liquid-popup STATIC src/LiquidPopup.cpp src/LiquidSurface.cpp src/LiquidDialog.cpp src/LiquidComboPopup.cpp src/LiquidButtons.cpp src/LiquidImagePicker.cpp)
set_target_properties(ukui-liquid-popup PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(ukui-liquid-popup PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(ukui-liquid-popup PUBLIC Qt5::Core Qt5::Gui Qt5::Widgets)
target_compile_features(ukui-liquid-popup PUBLIC cxx_std_17)
option(LIQUID_POPUP_BUILD_DEMO "Build standalone popup preview and tests" OFF)
if(LIQUID_POPUP_BUILD_DEMO)
  enable_testing()
  add_executable(liquid-popup-preview demo/main.cpp)
  target_link_libraries(liquid-popup-preview PRIVATE ukui-liquid-popup)
  add_executable(liquid-popup-test tests/test.cpp)
  target_link_libraries(liquid-popup-test PRIVATE ukui-liquid-popup)
  add_executable(liquid-dialog-test tests/dialog_test.cpp)
  target_link_libraries(liquid-dialog-test PRIVATE ukui-liquid-popup)
  add_executable(liquid-controls-test tests/controls_test.cpp)
  target_link_libraries(liquid-controls-test PRIVATE ukui-liquid-popup)
  add_executable(liquid-buttons-test tests/buttons_test.cpp)
  target_link_libraries(liquid-buttons-test PRIVATE ukui-liquid-popup)
  add_executable(liquid-image-picker-test tests/image_picker_test.cpp)
  target_link_libraries(liquid-image-picker-test PRIVATE ukui-liquid-popup)
  foreach(SCALE 1 1.25 1.5 2)
    add_test(NAME liquid-image-picker-${SCALE} COMMAND liquid-image-picker-test)
    set_tests_properties(liquid-image-picker-${SCALE} PROPERTIES TIMEOUT 25 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=${SCALE};QT_SCREEN_SCALE_FACTORS=1;QT_FONT_DPI=96")
    add_test(NAME liquid-dialog-${SCALE} COMMAND liquid-dialog-test)
    set_tests_properties(liquid-dialog-${SCALE} PROPERTIES TIMEOUT 40 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=${SCALE};QT_FONT_DPI=96")
    add_test(NAME liquid-controls-${SCALE} COMMAND liquid-controls-test)
    set_tests_properties(liquid-controls-${SCALE} PROPERTIES TIMEOUT 15 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=${SCALE};QT_FONT_DPI=96")
    add_test(NAME liquid-buttons-${SCALE} COMMAND liquid-buttons-test)
    set_tests_properties(liquid-buttons-${SCALE} PROPERTIES TIMEOUT 15 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=${SCALE};QT_SCREEN_SCALE_FACTORS=1;QT_FONT_DPI=96")
  endforeach()
  add_test(NAME liquid-image-picker-ukui COMMAND liquid-image-picker-test)
  set_tests_properties(liquid-image-picker-ukui PROPERTIES TIMEOUT 25 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=1.5;QT_SCREEN_SCALE_FACTORS=1;QT_FONT_DPI=96;UKUI_IMAGE_TEST_STYLE=ukui")
  add_test(NAME liquid-buttons-ukui COMMAND liquid-buttons-test)
  set_tests_properties(liquid-buttons-ukui PROPERTIES TIMEOUT 15 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=1.5;QT_SCREEN_SCALE_FACTORS=1;QT_FONT_DPI=96;UKUI_BUTTON_TEST_STYLE=ukui")
  add_test(NAME liquid-buttons-disabled COMMAND liquid-buttons-test)
  set_tests_properties(liquid-buttons-disabled PROPERTIES TIMEOUT 15 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;UKUI_LIQUID_POPUP=0")
  add_test(NAME liquid-dialog-disabled COMMAND liquid-dialog-test)
  add_test(NAME liquid-dialog-ukui COMMAND liquid-dialog-test)
  set_tests_properties(liquid-dialog-ukui PROPERTIES TIMEOUT 40 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=1.5;QT_FONT_DPI=96;UKUI_DIALOG_TEST_STYLE=ukui")
  set_tests_properties(liquid-dialog-disabled PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;UKUI_LIQUID_POPUP=0")
  add_test(NAME liquid-controls-disabled COMMAND liquid-controls-test)
  set_tests_properties(liquid-controls-disabled PROPERTIES TIMEOUT 15 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;UKUI_LIQUID_POPUP=0")
  find_program(LIQUID_XVFB xvfb-run)
  if(LIQUID_XVFB)
    add_test(NAME liquid-dialog-motion COMMAND ${LIQUID_XVFB} -a -s "-screen 0 2560x1440x24" $<TARGET_FILE:liquid-dialog-test>)
    set_tests_properties(liquid-dialog-motion PROPERTIES TIMEOUT 40 ENVIRONMENT "QT_SCALE_FACTOR=1.5")
  endif()
  add_test(NAME liquid-popup COMMAND liquid-popup-test)
  add_test(NAME liquid-popup-disabled COMMAND liquid-popup-test)
  set_tests_properties(liquid-popup-disabled PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;UKUI_LIQUID_POPUP=0")
  set_tests_properties(liquid-popup PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  foreach(SCALE 1 1.5 2)
    add_test(NAME liquid-popup-menu-text-${SCALE} COMMAND liquid-popup-test --menu-text-only)
    set_tests_properties(liquid-popup-menu-text-${SCALE} PROPERTIES
      ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=${SCALE};QT_SCREEN_SCALE_FACTORS=1;QT_FONT_DPI=96")
  endforeach()
endif()
